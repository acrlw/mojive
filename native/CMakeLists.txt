cmake_minimum_required(VERSION 3.24)
project(MojiveNativeProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(POLICY CMP0169)
    cmake_policy(SET CMP0169 OLD)
endif()
option(MOJIVE_BUILD_BGFX "Build the optional bgfx evaluation backend" ON)
option(MOJIVE_BUILD_SDL "Build the optional SDL3 GPU evaluation backend" OFF)
option(MOJIVE_BUILD_BINDINGS "Build isolated binding comparisons" OFF)
if(APPLE AND MOJIVE_BUILD_SDL)
    enable_language(OBJC)
endif()
add_library(mojive_render_contract INTERFACE)
target_include_directories(mojive_render_contract INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(WIN32)
    target_compile_definitions(mojive_render_contract INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
include(CTest)
if(MOJIVE_BUILD_BGFX OR MOJIVE_BUILD_SDL OR MOJIVE_BUILD_BINDINGS)
    include(cmake/Dependencies.cmake)
endif()
add_library(mojive_render_core STATIC src/contracts.cpp src/readback.cpp)
target_link_libraries(mojive_render_core PUBLIC mojive_render_contract)
option(MOJIVE_ENABLE_SANITIZERS "Instrument renderer code with address/undefined sanitizers" OFF)
if(MOJIVE_ENABLE_SANITIZERS)
    target_compile_options(mojive_render_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mojive_render_core PUBLIC -fsanitize=address,undefined)
endif()

if(MOJIVE_BUILD_BGFX)
    if(APPLE)
        set(shader_platform osx)
        set(shader_profile metal)
    elseif(WIN32)
        set(shader_platform windows)
        set(shader_profile s_5_0)
    else()
        set(shader_platform linux)
        set(shader_profile spirv)
    endif()
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shaders")
    set(shader_outputs)
    foreach(shader IN ITEMS vs_scene fs_color fs_data vs_ui fs_ui)
        if(shader MATCHES "^vs_")
            set(shader_type vertex)
        else()
            set(shader_type fragment)
        endif()
        set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/${shader}.bin")
        add_custom_command(OUTPUT "${output}"
            COMMAND mojive_shaderc -f "${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}.sc"
                -o "${output}" --type ${shader_type} --platform ${shader_platform}
                -p ${shader_profile} --varyingdef "${CMAKE_CURRENT_SOURCE_DIR}/shaders/varying.def.sc"
                -i "${BGFX_DIR}/src" -O 3
            DEPENDS mojive_shaderc "shaders/${shader}.sc" shaders/varying.def.sc VERBATIM)
        list(APPEND shader_outputs "${output}")
    endforeach()
    add_custom_target(mojive_probe_shaders ALL DEPENDS ${shader_outputs})
    add_library(mojive_backend_bgfx STATIC src/bgfx/renderer.cpp)
    target_link_libraries(mojive_backend_bgfx PUBLIC mojive_render_core PRIVATE bgfx)
    add_dependencies(mojive_backend_bgfx mojive_probe_shaders)
endif()
if(MOJIVE_BUILD_SDL)
    add_library(mojive_backend_sdl STATIC src/sdl/renderer.cpp)
    target_link_libraries(mojive_backend_sdl PUBLIC mojive_render_core PRIVATE SDL3::SDL3-static)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shaders")
    include(cmake/SdlShaders.cmake)
    add_executable(mojive_sdl_window_audit tools/sdl_window_audit.cpp)
    target_link_libraries(mojive_sdl_window_audit PRIVATE SDL3::SDL3-static)
endif()
if(MOJIVE_BUILD_BGFX OR MOJIVE_BUILD_SDL)
    set(native_tools probe benchmark gallery composition windows scene_capture)
    set(MOJIVE_MUJOCO_ROOT "" CACHE PATH "Optional MuJoCo distribution for the live runtime evaluation")
    if(MOJIVE_MUJOCO_ROOT)
        find_path(MOJIVE_MUJOCO_INCLUDE mujoco/mujoco.h PATHS "${MOJIVE_MUJOCO_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
        file(GLOB mujoco_libraries "${MOJIVE_MUJOCO_ROOT}/libmujoco*.dylib" "${MOJIVE_MUJOCO_ROOT}/libmujoco.so*" "${MOJIVE_MUJOCO_ROOT}/mujoco.lib")
        list(GET mujoco_libraries 0 MOJIVE_MUJOCO_LIBRARY)
        list(APPEND native_tools runtime)
    endif()
    foreach(tool IN LISTS native_tools)
        add_executable(mojive_native_${tool} tools/${tool}.cpp)
        if(MOJIVE_BUILD_BGFX)
            target_link_libraries(mojive_native_${tool} PRIVATE mojive_backend_bgfx)
            target_compile_definitions(mojive_native_${tool} PRIVATE MOJIVE_HAS_BGFX=1)
        endif()
        if(MOJIVE_BUILD_SDL)
            target_link_libraries(mojive_native_${tool} PRIVATE mojive_backend_sdl)
            target_compile_definitions(mojive_native_${tool} PRIVATE MOJIVE_HAS_SDL=1)
        endif()
    endforeach()
    target_link_libraries(mojive_native_gallery PRIVATE mojive_imgui glfw)
    target_link_libraries(mojive_native_windows PRIVATE glfw)
    if(TARGET mojive_native_runtime)
        target_include_directories(mojive_native_runtime PRIVATE "${MOJIVE_MUJOCO_INCLUDE}")
        target_link_libraries(mojive_native_runtime PRIVATE "${MOJIVE_MUJOCO_LIBRARY}")
        if(APPLE)
            # The Python wheel preserves the framework install name in its dylib.
            get_filename_component(mujoco_name "${MOJIVE_MUJOCO_LIBRARY}" NAME)
            set(mujoco_runtime "${CMAKE_CURRENT_BINARY_DIR}/lib/mujoco.framework/Versions/A")
            file(MAKE_DIRECTORY "${mujoco_runtime}")
            file(CREATE_LINK "${MOJIVE_MUJOCO_LIBRARY}" "${mujoco_runtime}/${mujoco_name}" SYMBOLIC)
            set_property(TARGET mojive_native_runtime APPEND PROPERTY BUILD_RPATH "${CMAKE_CURRENT_BINARY_DIR}/lib")
        endif()
    endif()
endif()

if(BUILD_TESTING)
    add_executable(mojive_native_contract_test tests/contract.cpp)
    target_link_libraries(mojive_native_contract_test PRIVATE mojive_render_core)
    add_test(NAME native_contract COMMAND mojive_native_contract_test)
    set_tests_properties(native_contract PROPERTIES LABELS cpu)
endif()

if(MOJIVE_BUILD_BINDINGS)
    include(cmake/Bindings.cmake)
endif()
